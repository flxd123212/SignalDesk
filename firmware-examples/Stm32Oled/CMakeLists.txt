cmake_minimum_required(VERSION 3.16)
set(CMAKE_SYSTEM_NAME Generic)
set(CMAKE_TRY_COMPILE_TARGET_TYPE STATIC_LIBRARY)
find_program(ARM_GCC arm-none-eabi-gcc REQUIRED)
set(CMAKE_C_COMPILER "${ARM_GCC}")
project(Stm32Oled C)
add_executable(oled.elf main.c)
target_compile_options(oled.elf PRIVATE -std=c99 -mcpu=cortex-m3 -mthumb -Os -g -ffreestanding)
target_link_options(oled.elf PRIVATE -mcpu=cortex-m3 -mthumb -nostdlib "-T${CMAKE_CURRENT_SOURCE_DIR}/memory.ld")

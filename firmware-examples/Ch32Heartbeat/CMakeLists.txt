cmake_minimum_required(VERSION 3.16)
set(CMAKE_SYSTEM_NAME Generic)
set(CMAKE_TRY_COMPILE_TARGET_TYPE STATIC_LIBRARY)
find_program(CH32_GCC NAMES riscv-none-elf-gcc riscv32-unknown-elf-gcc riscv64-unknown-elf-gcc riscv32-esp-elf-gcc REQUIRED)
set(CMAKE_C_COMPILER "${CH32_GCC}")
set(CMAKE_ASM_COMPILER "${CH32_GCC}")
project(Ch32Heartbeat C ASM)
add_executable(heartbeat.elf main.c start.S)
target_compile_options(heartbeat.elf PRIVATE -march=rv32ec -mabi=ilp32e -msmall-data-limit=0 -ffreestanding -Os -g)
target_link_options(heartbeat.elf PRIVATE -march=rv32ec -mabi=ilp32e -nostdlib -Wl,--no-relax "-T${CMAKE_CURRENT_SOURCE_DIR}/memory.ld")
